import pygame
import random

# Create your hero
class SuperHero:
    def __init__(self, name):
        self.name = name
        self.power = "Coding"
        self.level = 1
    
    def level_up(self):
        self.level += 1
        print(f"{self.name} is now Level {self.level}!")

# Start your adventure!
player = SuperHero("Alex")
player.level_up()
